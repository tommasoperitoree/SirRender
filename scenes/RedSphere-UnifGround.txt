    diffuse( uniform( (0, 0, 0) ) ) ,
    uniform( (1, 1, 1) )
)

    diffuse( uniform( (0.12, 0.2, 0.2) ) ) ,
    uniform( (0, 0, 0) )
)

    diffuse( uniform( (0.8, 0, 0) ) ) ,
    uniform( (0, 0, 0) )
)

sphere( skyMaterial, scaling( (15, 15, 15) ) )

plane( groundMaterial, identity )

sphere( sphereMaterial, scaling( (2, 2, 2) ) * translation( (0, 0, 0.8) ) )

camera( perspective, translation( (-6, 0, 5) ) * rotationY(30), 1.7777, 3.0 )
