float clock(150)

material wallLeftMaterial(
    diffuse( uniform( (1, 0, 0) ) ) ,
    uniform( (0, 0, 0) )
)

material wallRightMaterial(
    diffuse( uniform( (0, 1, 0) ) ) ,
    uniform( (0, 0, 0) )
)

material lampMaterial(
    diffuse( uniform((1, 1, 1) ) ) ,
    uniform( (100, 100, 100) )
)

material cubeMaterial(
    diffuse( uniform((1, 1, 0) ) ) ,
    uniform( (0, 0, 0) )
   )
material sphereMaterial(
   diffuse( uniform( (0, 1, 1) ) ) ,
   uniform( (0, 0, 0) )
)

material groundMaterial(
    diffuse( uniform( (1, 1, 1) ) ) ,
    uniform( (0, 0, 0) )
)

// side wall
plane( wallLeftMaterial, translation( (0, 5, 0) ) * rotationZ(90) * rotationY(90) )
plane( wallRightMaterial, translation( (0, -5, 0) ) * rotationZ(90) * rotationY(-90) )

// ground & back wall
plane( groundMaterial, identity ) // ground
plane( groundMaterial, translation( (5, 0, 0) ) * rotationY(90)) // back wall

// lamp
cube( lampMaterial, translation( (0, 0, 7) ) * scaling((1, 1, 0.5) ) )

// cube & sphere
cube( cubeMaterial, translation( (0, 2, 1) ) * scaling( (0.5, 0.5, 1) ) )
sphere( sphereMaterial,  translation( (0, -2, 0.5) ) * scaling( (0.5, 0.5, 0.5) ) )

camera( perspective,  translation( (-3,  0,  2.2) ),  1.77, 1 )
