float clock(0)

material skyMaterial(
    diffuse( uniform( (0, 0, 0) ) ),
    uniform( (10, 10, 10) )
)

material groundMaterial(
    diffuse( uniform( (0.3, 0.5, 0.3) ) ),
    uniform( (0, 0, 0) )
)

material cubeMaterial(
    diffuse( uniform( (1, 0, 0) ) ),
    uniform( (0, 0, 0) )
)

material sphereMaterial(
    diffuse( uniform( (0, 0, 1) ) ),
    uniform( (0, 0, 0) )
)

sphere( skyMaterial, scaling( (15, 15, 15) ) )

csg(
    union,
    cube( cubeMaterial, translation( (3, 3.5, 0) ) * scaling( (2, 2, 2) ) ),
    sphere( sphereMaterial, translation( (2.5, 3, 0.5) ) * scaling( (1.85, 1.85, 1.85) ) )
)

csg(
    difference,
    cube( cubeMaterial, translation( (-4, -3, 0) ) * scaling( (1.25, 1.25, 1.25) ) ),
    sphere( sphereMaterial, translation( (-4, -2.5, 0.5) ) * scaling( (1.25, 1.25, 1.25) ) )
)

csg(
    intersection,
    cube( cubeMaterial, translation( (-2, -7, 0) ) * scaling( (1.5, 1.5, 1.5) ) ),
    sphere( sphereMaterial, translation( (-2, -7, 0) ) * scaling( (1.75, 1.75, 1.75) ) )
)

camera(
    perspective,
    rotationZ( clock ) * translation( (-6, -4, 3) ) * rotationY(35),
    1.7777778,
    1.2
)
