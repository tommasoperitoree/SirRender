float clock(150)

material sphereMaterial(
    diffuse( uniform( (0.8, 0, 0) ) ) ,
    uniform( (0, 0, 0) )
)
material groundMaterial(
    diffuse( checkered( (1, 0.2, 0), //orange
                        (0, 0, 0), 4) ) ,
    uniform( (0, 0, 0) )
)

material sunMaterial(
    diffuse( uniform( (1, 0.4, 0) ) ) ,
    uniform( (1, 0.4, 0) )
)

material skyMaterial(
    diffuse( uniform( (0, 0, 0) ) ) ,
    uniform( (0.35, 0.75, 1) )
)

sphere( sunMaterial, translation( (-4.0, -3.5, 3.5) ) * scaling( (0.7, 0.7, 0.7) ) )
sphere( sphereMaterial, translation( (0, 0, 1) ) * scaling( (0.4, 0.4, 0.4) ) )
sphere( skyMaterial, scaling( (50, 50, 50) ) )
plane ( groundMaterial, identity )
camera( perspective, translation( (0, 0, 1) ) * rotationZ(-131), 1.777, 2.0 )
