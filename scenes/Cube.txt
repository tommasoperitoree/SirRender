float clock(150)

material skyMaterial(
    diffuse( uniform( (0, 0, 0) ) ) ,
    uniform( (0.35, 0.75, 1) )
)

material groundMaterial(
    diffuse( checkered( (1, 0.2, 0), //orange
                        (0, 0, 0), 4) ) ,
    uniform( (0, 0, 0) )
)

// blue cube
material cubeMaterial(
    diffuse( uniform( (0, 0, 0.8) ) ) ,
    uniform( (0, 0, 0) )
)


sphere( skyMaterial, scaling( (50, 50, 50) ) )
plane( groundMaterial, identity )

cube( cubeMaterial, translation( (0, 0, 0.5) ) * scaling( (0.5, 0.5, 0.5) ) )

camera( perspective, rotationZ(10) * rotationY(15) * translation( (-3, 0, 0.5) ), 1.777, 2.0 )
