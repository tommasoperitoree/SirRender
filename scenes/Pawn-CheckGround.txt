float clock(150)

material skyMaterial(
    diffuse( uniform( (0, 0, 0) ) ) ,
    uniform( (1.4, 3, 4) )
)

material groundMaterial(
    diffuse( checkered( (0.9, 0.9, 0.9), (0.2, 0.2, 0.2), 8) ) ,
    uniform( (0, 0, 0) )
)

material pawnMaterial(
    diffuse( uniform( (0.8, 0, 0) ) ) ,
    uniform( (0, 0, 0) )
)

sphere( skyMaterial, scaling( (20, 20, 20) ) )
plane( groundMaterial, identity )
mesh( pawnMaterial, file("scenes/pawn.obj"), identity )

camera( perspective, translation( (-7, 0, 3) ) * rotationY(10), 1.777, 2.0 )
