float clock(150)

material sky_material(
    diffuse(uniform((1, 1, 1))),
    uniform((1.4,3, 4))
)

material ground_material(
    diffuse(uniform((0,1,1))),
    uniform((0, 0, 0))
)

    diffuse(uniform((0.8, 0, 0))),
    uniform((0, 0, 0))
)

plane (ground_material, identity)

sphere(sphere_material, translation((0, 0, 1))*scaling((2,2,2)))

camera(perspective, rotation_z(30) * translation((-4, 0, 1)), 1.777, 2.0)
